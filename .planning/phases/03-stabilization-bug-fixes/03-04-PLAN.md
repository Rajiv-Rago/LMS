    For each route file, apply two changes:

    1. ADD validateObjectId at route entry (after auth, before DB call):
    ```typescript
    import { validateObjectId } from "@/lib/utils/validateObjectId";

    // At the start of each handler, before Course.findById():
    const invalidId = validateObjectId(id, "Course ID");
    if (invalidId) return invalidId;

    // For nested routes with multiple IDs (e.g., assignmentId, moduleId, lessonId):
    const invalidAssignmentId = validateObjectId(assignmentId, "Assignment ID");
    if (invalidAssignmentId) return invalidAssignmentId;
    ```

    2. REPLACE inline authorization with getCoursePermissions:
    ```typescript
    import { getCoursePermissions } from "@/lib/auth/coursePermissions";

    // Replace this pattern:
    // const isInstructor = course.instructor.toString() === user.userId;
    // const isAdmin = user.role === "admin";
    // const isEnrolled = course.enrolledStudents.some(...);
    // if (!isInstructor && !isAdmin && !isEnrolled) return 403;

    // With this:
    const perms = await getCoursePermissions(course, user);
    if (!perms.canView) {
      return NextResponse.json({ error: "Forbidden" }, { status: 403 });
    }
    // For write operations:
    if (!perms.canEdit) {
      return NextResponse.json({ error: "Forbidden" }, { status: 403 });
    }
    ```

    Authorization mapping per route:
    - GET (read data): require perms.canView
    - POST/PATCH/DELETE (modify data): require perms.canEdit
